using System.Threading.Tasks.Dataflow;
using static System.Runtime.InteropServices.JavaScript.JSType;

namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<int> list = new List<int>();

            for (int i = 1; i != 0;)
            {

                Console.Write("Digite um número para inserí-lo, ou 0 para terminar de adicionar: ");
                int numero = Convert.ToInt32(Console.ReadLine());
                i = numero;
                if (i != 0)
                {
                    list.Add(numero);
                }

            }
            Console.WriteLine();
            Console.WriteLine("Lista encerrada:");
            Console.WriteLine(string.Join(",", list));
            Console.WriteLine();

            int soma = 0;
            foreach (int i in list)
            {
                soma += i;

            }
            Console.WriteLine();
            Console.WriteLine($"A soma dos números é: {soma}");
            Console.WriteLine();

            int multiplicacao = 1;
            foreach (int i in list)
            {
                multiplicacao = i*multiplicacao;

            }
            Console.WriteLine();
            Console.WriteLine($"A multiplicação dos números é: {multiplicacao}");
            Console.WriteLine();

            list.Sort();
            Console.WriteLine($"{list[0]} é o menor número e {list[list.Count-1]} é o maior.");
        }

    }
}
