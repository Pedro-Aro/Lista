using System.Threading.Tasks.Dataflow;
using static System.Runtime.InteropServices.JavaScript.JSType;

namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<string> list = new List<string>();

            for (string i = "a"; i != "0";)
            {
                Console.Write("Digite uma palavra para inserí-la, ou 0 para terminar de adicionar: ");
                string inserido = (Console.ReadLine());

                if (inserido != "0")
                {
                    list.Add(inserido.ToString());
                }
                else
                {
                    i = inserido;
                }
            }
            Console.WriteLine();
            Console.WriteLine(string.Join(",", list));
            Console.WriteLine();

            Console.WriteLine("Digite a palavra para saber quantas vezes ela repetiu");
            string repetido = Console.ReadLine();
            list.RemoveAll(x => x != repetido);
            Console.WriteLine($"A palavra {repetido} se repetiu {list.Count} vezes");
        }

        }
}
